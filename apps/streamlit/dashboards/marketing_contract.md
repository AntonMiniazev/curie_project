# Calculations for Marketing dashboards
- average bill by month from fct_orders_sales.total_amount / count(distinct order_id);
- sales dynamics by month from sum(total_amount) grouped by month;
- monthly active clients from count(distinct client_id) grouped by month;
- client churn by month from dim_clients.churned, dim_clients.updated_at, and dim_clients.registration_date;
- new clients by registration month from dim_clients.registration_date;
- preferred-store client distribution from dim_clients.preferred_store_id -> dim_stores.store_id;
- total monthly sales-vs-budget from fct_orders_sales.order_date,total_amount and budget_orders_sales.budget_date,sales_amount_daily;
- store-level monthly actual-vs-budget from fct_orders_sales.store_id,total_amount and budget_orders_sales.store_id,sales_amount_daily;
- top products by sales amount from fct_order_product.line_sales_amount -> dim_products.product_name;
- top products by order frequency from count(distinct fct_order_product.order_id) -> dim_products.product_name;
- top categories by sales amount and order frequency through dim_products.category_name;
